SELECT
    CUSTOMER_BKEY,
    COUNT(DISTINCT CUSTOMER_CATEGORY_CODE)
        AS CATEGORY_COUNT,
    COUNT(DISTINCT INDUSTRY_CODE)
        AS INDUSTRY_COUNT,
    COUNT(DISTINCT NATIONALITY_CODE)
        AS NATIONALITY_COUNT,
    COUNT(DISTINCT IS_RESIDENT)
        AS RESIDENCY_COUNT
FROM SAMPLE_TABLE_2YEAR
WHERE CUSTOMER_BKEY IS NOT NULL
GROUP BY CUSTOMER_BKEY
HAVING COUNT(DISTINCT CUSTOMER_CATEGORY_CODE) > 1
    OR COUNT(DISTINCT INDUSTRY_CODE) > 1
    OR COUNT(DISTINCT NATIONALITY_CODE) > 1
    OR COUNT(DISTINCT IS_RESIDENT) > 1
FETCH FIRST 20 ROWS ONLY;
