CREATE OR REPLACE ATTRIBUTE DIMENSION PRODUCT_DIM
USING SAMPLE_TABLE_2YEAR
ATTRIBUTES (
    PRODUCT_CATEGORY_PARENT,
    PRODUCT_GROUP_CHILD,
    PRODUCT_TYPE_BKEY,
    PRODUCT_SUB_TYPE_BKEY
)
LEVEL PRODUCT_CATEGORY
    IS PRODUCT_CATEGORY_PARENT
LEVEL PRODUCT_GROUP
    IS PRODUCT_GROUP_CHILD
LEVEL PRODUCT_TYPE
    IS PRODUCT_TYPE_BKEY
LEVEL PRODUCT_SUB_TYPE
    IS PRODUCT_SUB_TYPE_BKEY;

    SELECT OBJECT_NAME, OBJECT_TYPE, STATUS
FROM USER_OBJECTS
WHERE OBJECT_NAME = 'PRODUCT_DIM';
