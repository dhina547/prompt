SELECT DIMENSION_NAME, LEVEL_NAME
FROM USER_DIM_LEVELS
WHERE DIMENSION_NAME = 'PRODUCT_DIM';


CREATE OR REPLACE HIERARCHY PRODUCT_HIER
USING PRODUCT_DIM
(
    PRODUCT_CATEGORY
        CHILD OF PRODUCT_GROUP
        CHILD OF PRODUCT_TYPE
        CHILD OF PRODUCT_SUB_TYPE
);

SELECT OBJECT_NAME, OBJECT_TYPE, STATUS
FROM USER_OBJECTS
WHERE OBJECT_NAME = 'PRODUCT_HIER';
