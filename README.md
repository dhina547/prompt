CREATE OR REPLACE ATTRIBUTE DIMENSION CUSTOMER_DIM
USING CUSTOMER_DIM_SOURCE
ATTRIBUTES (
    CUSTOMER_BKEY,
    CUSTOMER_CATEGORY_CODE,
    INDUSTRY_CODE,
    NATIONALITY_CODE,
    IS_RESIDENT,
    CUSTOMER_DOMICILE,
    BOI_CUSTOMER_CODE,
    CUSTOMER_SERIAL_CODE
)
LEVEL CUSTOMER_L1
    KEY CUSTOMER_BKEY
    MEMBER NAME TO_CHAR(CUSTOMER_BKEY);


    SELECT OBJECT_NAME, OBJECT_TYPE, STATUS
FROM USER_OBJECTS
WHERE OBJECT_NAME = 'CUSTOMER_DIM';
