SELECT
    PRODUCT_CATEGORY_PARENT,
    PRODUCT_GROUP_CHILD,
    PRODUCT_TYPE_BKEY,
    PRODUCT_SUB_TYPE_BKEY,
    INSTRUMENT_CODE,
    COUNT(*) AS ROW_COUNT
FROM SAMPLE_TABLE_2YEAR
GROUP BY
    PRODUCT_CATEGORY_PARENT,
    PRODUCT_GROUP_CHILD,
    PRODUCT_TYPE_BKEY,
    PRODUCT_SUB_TYPE_BKEY,
    INSTRUMENT_CODE
