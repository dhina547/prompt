SELECT
    COA_BALANCE_AMOUNT_LCY,
    PROVISION_AMOUNT_LCY,
    ACCOUNTING_DEAL_DATE,
    COA_BALANCE_AMOUNT,
    MATURITY_DATE,
    DEAL_DATE,
    SOURCE_SYSTEM_TIMESTAMP,
    STRIKE_PRICE,
    PREMIUM_AMOUNT,
    PREMIUM_RATE,
    COF_RATE,
    PRINCIPAL_AMOUNT,
    OUTSTANDING_QUANTITY,
    MARKET_VALUE,
    CLOSE_DATE,
    REPORT_CODE,
    INTEREST_FREQ,
    QUANTITY,
    PURCHASED_INTEREST,
    OUTSTANDING_PRINCIPAL_AMOUNT,
    OUTSTANDING_BOOK_VALUE,
    MARKET_PRICE,
    ACCOUNT_NATURE,
    LIMIT_UNDRAWN_AMT_LCY,
    LIMIT_UTILIZED_AMT_LCY,
    LIMIT_UTIL_CURRENCY_CODE,
    LIMIT_AMT_LCY,
    ACCOUNTING_INTEREST_RATE,
    ACCOUNTING_MATURITY_DATE,
    LIQUIDATION_BALANCE_AMOUNT,
    COA_CURRENCY_CODE,
    BUSINESS_DATE,
    BOOK_VALUE,
    ACCRUED_INTEREST,
    ISSUE_DATE,
    PREMIUM_CURRENCY_CODE,
    VALUE_DATE,
    TRADE_CLASS,
    HEDGING_TRADING_IND,
    LIMIT_INITIAL_AMT,
    LIMIT_EXPIRY_DATE,
    LIMIT_APPROVED_AMT,
    TRADE_MASTER_BKEY,
    CUSTOMER_BKEY,
    USER_CODE,
    CUSTOMER_BRANCH_CODE,
    BANK_IDENTIFIER_CODE,
    CUSTOMER_STATUS_CODE,
    CUSTOMER_SERIAL_CODE,
    CUSTOMER_NAME,
    CITY,
    NATIONAL_ID
FROM (
    SELECT t.*,
           ROW_NUMBER() OVER (
               PARTITION BY BUSINESS_DATE
               ORDER BY ROWID
           ) AS RN
    FROM LEDGER_MASTER_MONTHLY_HIST t
    WHERE BUSINESS_DATE IN (
        DATE '2025-12-31',
        DATE '2025-10-31',
        DATE '2025-06-30',
        DATE '2025-11-30',
        DATE '2026-06-30',
        DATE '2025-08-31',
        DATE '2025-07-31',
        DATE '2026-07-31',
        DATE '2026-03-31',
        DATE '2026-05-31',
        DATE '2026-04-30',
        DATE '2026-01-31',
        DATE '2026-02-28'
    )
)
WHERE
      (BUSINESS_DATE IN (
          DATE '2025-12-31',
          DATE '2025-10-31'
      ) AND RN <= 3847)
   OR (BUSINESS_DATE IN (
          DATE '2025-06-30',
          DATE '2025-11-30',
          DATE '2026-06-30',
          DATE '2025-08-31',
          DATE '2025-07-31',
          DATE '2026-07-31',
          DATE '2026-03-31',
          DATE '2026-05-31',
          DATE '2026-04-30',
          DATE '2026-01-31',
          DATE '2026-02-28'
      ) AND RN <= 3846);
