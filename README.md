SELECT
    PRODUCT_GROUP_CHILD,
    COUNT(DISTINCT PRODUCT_CATEGORY_PARENT) AS CATEGORY_COUNT
FROM SAMPLE_TABLE_2YEAR
WHERE PRODUCT_GROUP_CHILD IS NOT NULL
GROUP BY PRODUCT_GROUP_CHILD
HAVING COUNT(DISTINCT PRODUCT_CATEGORY_PARENT) > 1;


SELECT
    PRODUCT_TYPE_BKEY,
    COUNT(DISTINCT PRODUCT_GROUP_CHILD) AS GROUP_COUNT
FROM SAMPLE_TABLE_2YEAR
WHERE PRODUCT_TYPE_BKEY IS NOT NULL
GROUP BY PRODUCT_TYPE_BKEY
HAVING COUNT(DISTINCT PRODUCT_GROUP_CHILD) > 1;

SELECT
    PRODUCT_SUB_TYPE_BKEY,
    COUNT(DISTINCT PRODUCT_TYPE_BKEY) AS TYPE_COUNT
FROM SAMPLE_TABLE_2YEAR
WHERE PRODUCT_SUB_TYPE_BKEY IS NOT NULL
GROUP BY PRODUCT_SUB_TYPE_BKEY
HAVING COUNT(DISTINCT PRODUCT_TYPE_BKEY) > 1;
