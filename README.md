CREATE OR REPLACE VIEW CUSTOMER_DIM_SOURCE AS
SELECT
    CUSTOMER_BKEY,
    MAX(CUSTOMER_CATEGORY_CODE) AS CUSTOMER_CATEGORY_CODE,
    MAX(INDUSTRY_CODE)          AS INDUSTRY_CODE,
    MAX(NATIONALITY_CODE)       AS NATIONALITY_CODE,
    MAX(IS_RESIDENT)            AS IS_RESIDENT,
    MAX(CUSTOMER_DOMICILE)      AS CUSTOMER_DOMICILE,
    MAX(BOI_CUSTOMER_CODE)      AS BOI_CUSTOMER_CODE,
    MAX(CUSTOMER_SERIAL_CODE)   AS CUSTOMER_SERIAL_CODE
FROM SAMPLE_TABLE_2YEAR
WHERE CUSTOMER_BKEY IS NOT NULL
GROUP BY CUSTOMER_BKEY;






SELECT
    COUNT(*) AS CUSTOMER_ROWS,
    COUNT(DISTINCT CUSTOMER_BKEY) AS DISTINCT_KEYS
FROM CUSTOMER_DIM_SOURCE;
