SELECT
    CUSTOMER_BKEY,
    COUNT(DISTINCT CUSTOMER_DOMICILE) AS DOMICILE_COUNT,
    COUNT(DISTINCT BOI_CUSTOMER_CODE) AS BOI_CODE_COUNT,
    COUNT(DISTINCT CHART_OF_ACCOUNT_BKEY) AS CHART_ACCOUNT_COUNT,
    COUNT(DISTINCT CUSTOMER_SERIAL_CODE) AS SERIAL_CODE_COUNT
FROM SAMPLE_TABLE_2YEAR
WHERE CUSTOMER_BKEY = 4000002022
GROUP BY CUSTOMER_BKEY;
