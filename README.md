CREATE OR REPLACE ATTRIBUTE DIMENSION PRODUCT_DIM
USING SAMPLE_TABLE_2YEAR
ATTRIBUTES
(
    PRODUCT_CATEGORY_PARENT,
    PRODUCT_GROUP_CHILD,
    PRODUCT_TYPE_BKEY,
    PRODUCT_SUB_TYPE_BKEY
)
LEVEL PRODUCT_CATEGORY
    KEY PRODUCT_CATEGORY_PARENT
    SKIP WHEN NULL
LEVEL PRODUCT_GROUP
    KEY (PRODUCT_CATEGORY_PARENT, PRODUCT_GROUP_CHILD)
    SKIP WHEN NULL
LEVEL PRODUCT_TYPE
    KEY (PRODUCT_GROUP_CHILD, PRODUCT_TYPE_BKEY)
    SKIP WHEN NULL
LEVEL PRODUCT_SUB_TYPE
    KEY (PRODUCT_TYPE_BKEY, PRODUCT_SUB_TYPE_BKEY)
    SKIP WHEN NULL;


    CREATE OR REPLACE HIERARCHY PRODUCT_HIER
USING PRODUCT_DIM
(
    PRODUCT_SUB_TYPE
        CHILD OF PRODUCT_TYPE
        CHILD OF PRODUCT_GROUP
        CHILD OF PRODUCT_CATEGORY
);


SELECT OBJECT_NAME, OBJECT_TYPE, STATUS
FROM USER_OBJECTS
WHERE OBJECT_NAME IN ('PRODUCT_DIM', 'PRODUCT_HIER');
